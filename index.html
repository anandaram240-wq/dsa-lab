#include <stdio.h>
#include <stdlib.h>
#define max 4

int s[max], top = -1;

void push(int s[], int val) {
    if (top == max-1) { printf("STACK OVERFLOW\n"); return; }
    s[++top] = val;
    printf("%d inserted\n", val);
}

void pop(int s[]) {
    if (top == -1) { printf("STACK UNDERFLOW\n"); return; }
    printf("%d deleted\n", s[top--]);
}

void display(int s[]) {
    if (top == -1) { printf("EMPTY\n"); return; }
    for(int i=top; i>=0; i--) printf("%d\n", s[i]);
}

void main() {
    int val, option;
    do {
        printf("1.push 2.pop 3.display 4.exit\n");
        scanf("%d", &option);
        switch(option) {
            case 1: scanf("%d",&val); push(s,val); break;
            case 2: pop(s); break;
            case 3: display(s); break;
            case 4: exit(0);
        }
    } while(option != 4);
}

void insert(int q[], int val) {

#include <string.h>
#include <stdlib.h>
#include <ctype.h>
#define max 100

char s[max]; int top = -1;

void push(char s[], char val) {
    if(top==max-1) { printf("OVERFLOW\n"); return; }
    s[++top] = val;
}
char pop(char s[]) {
    if(top==-1) { printf("UNDERFLOW\n"); return ' '; }
    return s[top--];
}
int getPriority(char op) {
    if(op=='^'||op=='$') return 2;
    if(op=='/'||op=='*'||op=='%') return 1;
    if(op=='+'||op=='-') return 0;
    return -1;
}

void InfixToPostfix(char infix[], char postfix[]) {
    int i=0, j=0; char temp;
    while(infix[i] != '\0') {
        if(infix[i]=='(') { push(s,infix[i]); i++; }
        else if(infix[i]==')') {
            while(top!=-1 && s[top]!='(') postfix[j++]=pop(s);
            temp=pop(s); // discard '('
            i++;
        }
        else if(isdigit(infix[i])||isalpha(infix[i])) { postfix[j++]=infix[i++]; }
        else {
            while(top!=-1 && s[top]!='(' && getPriority(s[top])>=getPriority(infix[i]))
                postfix[j++]=pop(s);
            push(s,infix[i++]);
        }
    }
    while(top!=-1) postfix[j++]=pop(s);
    postfix[j]='\0';
}

int main() {
    char infix[100], postfix[100];
    printf("Enter infix expression\n");
    scanf("%s", infix);
    InfixToPostfix(infix, postfix);
    printf("Postfix: %s\n", postfix);
    return 0;
}

#include <stdio.h>
#include <ctype.h>
#include <math.h>
#define max 100

double s[max]; int top = -1; // NOTE: double stack!

void push(double s[], double val) { s[++top]=val; }
double pop(double s[]) { return s[top--]; }

double evaluate(char postfix[]) {
    int i=0;
    double op1, op2, value;
    while(postfix[i] != '\0') {
        if(isdigit(postfix[i])) {
            push(s, (double)(postfix[i]-'0')); // digit→number
        } else {
            op2 = pop(s); // pop op2 FIRST!
            op1 = pop(s); // then op1
            switch(postfix[i]) {
                case '+': value=op1+op2; break;
                case '-': value=op1-op2; break;
                case '*': value=op1*op2; break;
                case '/': value=op1/op2; break;
                case '%': value=(int)op1%(int)op2; break;
                case '^':
                case '$': value=pow(op1,op2); break;
            }
            push(s, value);
        }
        i++;
    }
    return pop(s);
}

void main() {
    char postfix[100];
    printf("Enter postfix expression\n");
    scanf("%s", postfix);
    printf("Result = %lf\n", evaluate(postfix));
}

#include <stdlib.h>

struct node { int info; struct node *link; };
typedef struct node *NODE;
NODE first = NULL;

NODE insert(NODE first) {
    NODE temp; int item;
    printf("Enter value\n"); scanf("%d",&item);
    temp = (NODE)malloc(sizeof(struct node));
    temp->info = item;
    temp->link = first;  // point to old first
    first = temp;        // new node becomes first
    printf("%d inserted\n", item);
    return first;
}

NODE delete_sp(NODE first) {
    NODE cur, prev; int key;
    printf("Enter value to delete\n"); scanf("%d",&key);
    if(first==NULL) { printf("Empty\n"); return first; }
    if(key==first->info) { cur=first; first=first->link; free(cur); return first; }
    cur=first; prev=NULL;
    while(cur!=NULL && cur->info!=key) { prev=cur; cur=cur->link; }
    if(cur==NULL) { printf("Not found\n"); return first; }
    prev->link = cur->link;
    printf("%d deleted\n", cur->info);
    free(cur); return first;
}

void display(NODE first) {
    NODE cur = first;
    if(first==NULL) { printf("Empty\n"); return; }
    while(cur!=NULL) { printf("%d\t", cur->info); cur=cur->link; }
    printf("\n");
}

NODE first = NULL;

NODE insertRear(NODE first) {
    NODE temp, cur; int item;
    printf("Enter value\n"); scanf("%d",&item);
    temp=(NODE)malloc(sizeof(struct node));
    temp->info=item; temp->link=temp; // point to itself
    if(first==NULL) { first=temp; return first; }
    cur=first;
    while(cur->link!=first) cur=cur->link; // find last
    cur->link=temp; temp->link=first; // link at rear
    printf("%d inserted\n",item);
    return first;
}

NODE deleteFront(NODE first) {
    NODE cur;
    if(first==NULL) { printf("Empty\n"); return first; }
    if(first->link==first) { // only 1 node
        printf("%d deleted\n",first->info);
        free(first); return NULL;
    }
    cur=first;
    while(cur->link!=first) cur=cur->link; // find last
    cur->link=first->link; // last points to 2nd
    printf("%d deleted\n",first->info);
    free(first); first=cur->link; return first;
}

void display(NODE first) {
    NODE cur;

    cur=first;
    while(cur->link!=first) { printf("%d\t",cur->info); cur=cur->link; }

struct node { int info; struct node *llink, *rlink; };
typedef struct node *NODE;
NODE first = NULL; // llink=left(prev) rlink=right(next)

NODE insert_begin(NODE first) {
    NODE temp; int item;
    printf("Enter value\n"); scanf("%d",&item);
    temp=(NODE)malloc(sizeof(struct node));
    temp->info=item; temp->llink=NULL; temp->rlink=NULL;
    if(first==NULL) { first=temp; return first; }
    first->llink=temp;   // old first's left = new
    temp->rlink=first;   // new right = old first
    first=temp;          // update first
    printf("%d inserted\n",item);
    return first;
}

NODE delete_begin(NODE first) {
    NODE cur;
    if(first==NULL) { printf("Empty\n"); return first; }
    cur=first;
    first=first->rlink;          // move first to next
    if(first!=NULL) first->llink=NULL; // fix left pointer
    printf("%d deleted\n",cur->info);
    free(cur); return first;
}

NODE display(NODE first) {
    NODE cur=first;
    if(first==NULL) { printf("Empty\n"); return first; }
    while(cur!=NULL) { printf("%d\t",cur->info); cur=cur->rlink; }
    printf("\n"); return first;
}

int main() {
    int option;
    do {
        printf("1.Insert 2.Delete 3.Display 4.Exit\n");
        scanf("%d",&option);
        switch(option) {
            case 1: first=insert_begin(first); break;
            case 2: first=delete_begin(first); break;
            case 3: first=display(first); break;
            case 4: break;
        }
    } while(option!=4);
    return 0;
}

Exp 5a – Binary Search (Recursive)

// THE WHOLE TRICK: call itself with smaller range
int search(int key, int a[], int low, int high) {
    if (low > high) return -1;  // NOT FOUND
    mid = (low + high) / 2;
    if (key == a[mid]) return mid;  // FOUND!
    if (key < a[mid])
        return search(key, a, low, mid-1);
    else
        return search(key, a, mid+1, high);
}
// Call: search(key, a, 0, n-1)
🧠 Memory: Low>High=not found, found=return mid, smaller=go left (low to mid-1), bigger=go right (mid+1 to high)

Exp 5b – Tower of Hanoi

// 3 pegs: src, dest, temp
void move(int n, char src, char dest, char temp) {
    if (n == 1) {
        printf("Move disc from %c to %c\n", src, dest);
        return;
    }
    move(n-1, src, temp, dest);  // move n-1 to temp
    move(1, src, dest, temp);    // move bottom
    move(n-1, temp, dest, src); // move n-1 to dest
}
// Call: move(n, 'A', 'C', 'B')
🧠 Memory: n=1 is the stop. 3 calls: move small ones aside, move big disc, bring small ones back.

    NODE temp, cur, prev; int item;
    printf("Enter value\n"); scanf("%d",&item);
    temp=(NODE)malloc(sizeof(struct node));
    temp->info=item; temp->llink=NULL; temp->rlink=NULL;
    if(root==NULL) { root=temp; return root; }
    prev=NULL; cur=root;
    while(cur!=NULL) {
        prev=cur;
        if(item < cur->info) cur=cur->llink;  // go left
        else cur=cur->rlink;                   // go right
    }
    if(item < prev->info) prev->llink=temp;
    else prev->rlink=temp;
    return root;
}

void preorder(NODE root) {  // Root→Left→Right
    if(root==NULL) return;
    printf("%d\t",root->info);
    preorder(root->llink); preorder(root->rlink);
}
void inorder(NODE root) {   // Left→Root→Right (gives SORTED output!)
    if(root==NULL) return;
    inorder(root->llink);
    printf("%d\t",root->info);
    inorder(root->rlink);
}
void postorder(NODE root) { // Left→Right→Root
    if(root==NULL) return;
    postorder(root->llink); postorder(root->rlink);
    printf("%d\t",root->info);
}

int main() {
    int n; printf("Enter number of nodes\n"); scanf("%d",&n);
    while(n--) root=insert(root);
    printf("\nPreorder: "); preorder(root);
    printf("\nInorder:  "); inorder(root);
    printf("\nPostorder:"); postorder(root);
    printf("\n"); return 0;
}

/* ===== EXP 8: SINGLY LL — Delete by KEY ===== */
NODE singly_delete(NODE first) {
    NODE cur, prev; int key;
    printf("Enter key to delete: "); scanf("%d", &key);
    if (first == NULL) { printf("List is empty\n"); return first; }
    if (key == first->info) {
        cur = first; first = first->link;
        printf("%d deleted\n", cur->info); free(cur); return first;
    }
    cur = first; prev = NULL;
    while (cur != NULL && cur->info != key) { prev = cur; cur = cur->link; }
    if (cur == NULL) { printf("Key not found\n"); return first; }
    prev->link = cur->link;
    printf("%d deleted\n", cur->info); free(cur); return first;
}

NODE ordered_insert(NODE ofirst) {
    NODE temp, cur, prev; int item;
    printf("Enter value: "); scanf("%d", &item);
    temp = (NODE)malloc(sizeof(struct node));
    temp->info = item; temp->link = NULL;

    while (cur != NULL && item > cur->info) { prev = cur; cur = cur->link; }

    cur->link = temp; temp->link = cfirst;

}

NODE circular_delete(NODE cfirst) {

    while (cur->link != cfirst) cur = cur->link;
    cur->link = cfirst->link;
    printf("%d deleted (circular)\n", cfirst->info);
    free(cfirst); cfirst = cur->link; return cfirst;
}
